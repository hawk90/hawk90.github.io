% Standard R-2R ladder: each bit Dn drives a 2R leg into node Nn, nodes are
% chained by R, the LSB end is terminated by 2R to GND, and the MSB node is
% the output. Unloaded: V_out = V_REF * code / 256.
\begin{tikzpicture}[blog,
  wire/.style={draw=inkc, thick},
  resV/.style={draw=inkc, thick, fill=white, minimum width=2.2mm, minimum height=7mm, inner sep=0pt},
  resH/.style={draw=inkc, thick, fill=white, minimum width=7mm, minimum height=2.2mm, inner sep=0pt},
  dot/.style={circle, fill=inkc, inner sep=1.4pt},
  lbl/.style={font=\sffamily\scriptsize, text=inkc}]

\node[font=\sffamily\bfseries\small, text=inkc] at (5.25, 5.0) {R-2R Ladder DAC (8비트)};

% node x positions: D0, D1, D2, (dots), D6, D7
\foreach \n/\x in {0/1.5, 1/3.0, 2/4.5, 6/7.0, 7/8.5} {
  \coordinate (N\n) at (\x, 2.0);
}

% ladder spine with series R between neighbouring nodes
\draw[wire] (N0) -- (N2);
\draw[wire] (N2) -- ($(N2)+(0.6,0)$);
\node[lbl] at (5.75, 2.0) {$\cdots$};
\draw[wire] ($(N6)+(-0.6,0)$) -- (N6) -- (N7) -- (9.8, 2.0);
\foreach \a/\b in {N0/N1, N1/N2, N6/N7} {
  \node[resH] (r\a) at ($(\a)!0.5!(\b)$) {};
  \node[lbl, below=1.5mm] at (r\a) {R};
}

% bit legs: Dn -- 2R -- Nn
\foreach \n in {0,1,2,6,7} {
  \draw[wire] (N\n) -- ++(0, 2.0) coordinate (T\n);
  \node[resV] (l\n) at ($(N\n)+(0,1.0)$) {};
  \node[lbl, right=1.5mm] at (l\n) {2R};
  \node[lbl, above=1mm] at (T\n) {D\n};
  \node[dot] at (N\n) {};
}
\node[lbl] at (5.75, 4.15) {$\cdots$};

% LSB termination: N0 -- 2R -- GND
\draw[wire] (N0) -- ++(0, -1.6) coordinate (G);
\node[resV] (term) at ($(N0)+(0,-0.8)$) {};
\node[lbl, right=1.5mm] at (term) {2R (종단)};
\draw[wire] ($(G)+(-0.3,0)$) -- ($(G)+(0.3,0)$);
\draw[wire] ($(G)+(-0.2,-0.1)$) -- ($(G)+(0.2,-0.1)$);
\draw[wire] ($(G)+(-0.1,-0.2)$) -- ($(G)+(0.1,-0.2)$);
\node[lbl, below=2mm] at (G) {GND};

% output at the MSB end
\node[dot] at (9.8, 2.0) {};
\node[lbl, right=1mm] at (9.8, 2.0) {V\textsubscript{out}};

\node[font=\sffamily\scriptsize, text=mutec, align=center] at (6.6, 0.0)
  {D\textsubscript{n} = 1이면 2R이 V\textsubscript{REF}에, 0이면 GND에 연결됩니다\\[2pt]
   V\textsubscript{out} = V\textsubscript{REF} × code / 256 (MSB인 D7이 출력에 가장 가깝습니다)};

\end{tikzpicture}
\end{document}
