% One start bit = 16 samples (OVER8 = 0), RM0433 48.5.6
\def\bw{9.6}
\def\sw{0.6}  % \bw / 16

% Falling edge at 0, start bit, then D0 high
\draw[sig] (-1.5, 1) -- (0, 1) -- (0, 0) -- (\bw, 0) -- (\bw, 1) -- (\bw + 1.5, 1);

% 16 sample ticks across the start bit, numbered
\foreach \i in {1,...,16} {
  \draw[tick] (\i*\sw - \sw/2, -0.15) -- (\i*\sw - \sw/2, 0);
}
\foreach \i in {1, 3, 5, 7, 8, 9, 10, 16} {
  \node[sublbl, anchor=south] at (\i*\sw - \sw/2, 0.05) {\i};
}

% Group 1: samples 3, 5, 7
\foreach \s in {3, 5, 7} {
  \fill[tealfill, draw=tealbord, line width=0.3pt]
    (\s*\sw - \sw/2 - 0.08, -0.2) rectangle (\s*\sw - \sw/2 + 0.08, 0.0);
}
% Group 2: samples 8, 9, 10
\foreach \s in {8, 9, 10} {
  \fill[hifill, draw=hibord, line width=0.3pt]
    (\s*\sw - \sw/2 - 0.08, -0.2) rectangle (\s*\sw - \sw/2 + 0.08, 0.0);
}

% Group braces
\draw[decorate, decoration={brace, amplitude=3pt, mirror}, draw=tealbord, line width=0.4pt]
  (3*\sw - \sw/2 - 0.1, -0.3) -- (7*\sw - \sw/2 + 0.1, -0.3);
\node[sublbl, text=tealbord, anchor=north] at (5*\sw - \sw/2, -0.45) {확인 1: 3·5·7};

\draw[decorate, decoration={brace, amplitude=3pt, mirror}, draw=hibord, line width=0.4pt]
  (8*\sw - \sw/2 - 0.1, -0.3) -- (10*\sw - \sw/2 + 0.1, -0.3);
\node[sublbl, text=hibord, anchor=north] at (9*\sw - \sw/2, -0.45) {확인 2: 8·9·10};

% Rule
\node[sublbl, text=inkc, anchor=north west, align=left] at (-1.5, -1.05)
  {두 묶음 모두 셋 중 둘 이상이 0 $\to$ start bit로 받음\\[2pt]
   어느 묶음이든 셋 중 둘만 0 $\to$ 받고 NE를 세움\\[2pt]

\node[lbl, anchor=east] at (-1.6, 1) {idle};
