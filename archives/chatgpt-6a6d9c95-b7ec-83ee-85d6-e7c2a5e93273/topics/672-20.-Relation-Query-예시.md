# 20. Relation Query 예시

```ts
export function getPrerequisites(
  documentId: string,
  graph: RelationGraph,
  manifest: ContentManifest,
): ContentDocument[] {
  const ids =
    graph.prerequisitesByDocument
      .get(documentId) ?? new Set();

  return [...ids]
    .map((id) => manifest.byId.get(id))
    .filter(isDefined);
}
```
