# 43. Draft 정책

```ts
function getDraftDecision(
  environment: BuildEnvironment,
): PublicationDecision {
  if (environment === "production") {
    return {
      render: false,

      index: false,
      follow: false,

      includeInSearch: false,
      includeInSitemap: false,
      includeInRss: false,
      includeInTopicLists: false,

      canBeFeatured: false,
      adsEligible: false,

      searchRankMultiplier: 0,
    };
  }

  return {
    render: true,

    index: false,
    follow: false,

    includeInSearch: false,
    includeInSitemap: false,
    includeInRss: false,
    includeInTopicLists: false,

    canBeFeatured: false,
    adsEligible: false,

    searchRankMultiplier: 0,
  };
}
```
