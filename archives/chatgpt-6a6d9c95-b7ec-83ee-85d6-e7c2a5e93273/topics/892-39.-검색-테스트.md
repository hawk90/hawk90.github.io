# 39. 검색 테스트

```ts
describe("golden queries", () => {
  it.each(GOLDEN_QUERIES)(
    "$query",
    ({ query, mustInclude }) => {
      const results =
        searchEngine.search(query);

      const resultIds =
        results.map(
          (result) =>
            result.documentId,
        );

      for (const id of mustInclude) {
        expect(resultIds)
          .toContain(id);
      }
    },
  );
});
```
