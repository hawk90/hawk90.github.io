# 9. 관계 그래프는 데이터베이스가 필요하지 않다

문서 수가 수백 개라면 `Map`과 `Set`이면 충분하다.

```ts
export interface RelationGraph {
  prerequisitesByDocument:
    ReadonlyMap<string, ReadonlySet<string>>;

  dependentsByDocument:
    ReadonlyMap<string, ReadonlySet<string>>;

  nextByDocument:
    ReadonlyMap<string, ReadonlySet<string>>;

  relatedByDocument:
    ReadonlyMap<string, ReadonlySet<string>>;
}
```
