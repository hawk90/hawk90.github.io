# 74. 관계 Graph 테스트

```ts
it("creates reverse prerequisite links", () => {
  const manifest =
    createContentManifest([
      createDocumentFixture({
        id: "foundation",
      }),

      createDocumentFixture({
        id: "advanced",
        relations: {
          prerequisites: ["foundation"],
          next: [],
          related: [],
        },
      }),
    ]);

  const graph =
    buildRelationGraph(manifest);

  expect(
    graph.dependentsByDocument
      .get("foundation"),
  ).toContain("advanced");
});
```
