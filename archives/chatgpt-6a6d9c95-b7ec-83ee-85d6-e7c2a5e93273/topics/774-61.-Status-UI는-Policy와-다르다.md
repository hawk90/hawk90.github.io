# 61. Status UI는 Policy와 다르다

Publication Policy가 `index=false`라고 해서 자동으로 경고 배너 문구를 결정하면 안 된다.

별도 UI 모델:

```ts
export interface StatusPresentation {
  showNotice: boolean;
  tone:
    | "neutral"
    | "warning"
    | "historical";

  title?: string;
  message?: string;
}
```

예:

```ts
export function getStatusPresentation(
  document: ContentDocument,
): StatusPresentation {
  switch (document.status) {
    case "current":
      return {
        showNotice: false,
        tone: "neutral",
      };

    case "needs-review":
      return {
        showNotice: true,
        tone: "warning",
        title: "검토가 필요한 문서",
        message:
          "일부 환경이나 버전 정보가 " +
          "현재와 다를 수 있습니다.",
      };

    case "historical":
      return {
        showNotice: true,
        tone: "historical",
        title: "과거 환경을 다루는 문서",
        message:
          "이 문서는 특정 시점의 " +
          "환경과 동작을 기록합니다.",
      };

    case "superseded":
      return {
        showNotice: true,
        tone: "warning",
        title: "새 문서로 대체됨",
        message:
          "현재 내용은 대체 문서를 " +
          "먼저 확인하세요.",
      };

    case "archived":
      return {
        showNotice: true,
        tone: "neutral",
        title: "보존 문서",
        message:
          "일반 학습 경로에서는 " +
          "사용하지 않는 기록입니다.",
      };
  }
}
```
