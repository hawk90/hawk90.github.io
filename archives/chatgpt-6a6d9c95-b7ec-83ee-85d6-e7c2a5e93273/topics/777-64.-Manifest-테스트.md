# 64. Manifest 테스트

## 중복 ID

```ts
it("rejects duplicate document IDs", () => {
  const documents = [
    createDocumentFixture({
      id: "duplicate",
    }),
    createDocumentFixture({
      id: "duplicate",
      slug: "other",
    }),
  ];

  expect(() =>
    createContentManifest(documents)
  ).toThrow(
    /Duplicate content ID/,
  );
});
```

## Relation target

```ts
it("rejects missing relation targets", () => {
  const documents = [
    createDocumentFixture({
      id: "article-a",
      relations: {
        prerequisites: [
          "missing-article",
        ],
        next: [],
        related: [],
      },
    }),
  ];

  expect(() =>
    createContentManifest(documents)
  ).toThrow(
    /does not exist/,
  );
});
```
