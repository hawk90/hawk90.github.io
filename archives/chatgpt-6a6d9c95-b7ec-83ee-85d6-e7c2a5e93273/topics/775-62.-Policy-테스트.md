# 62. Policy 테스트

## Current

```ts
it("allows current documents everywhere", () => {
  const document =
    createDocumentFixture({
      status: "current",
    });

  const decision =
    getPublicationDecision(
      document,
      testContext,
    );

  expect(decision.render).toBe(true);
  expect(decision.index).toBe(true);
  expect(
    decision.includeInSearch,
  ).toBe(true);
  expect(
    decision.canBeFeatured,
  ).toBe(true);
});
```

## Needs Review

```ts
it("does not allow needs-review documents to be featured", () => {
  const document =
    createDocumentFixture({
      status: "needs-review",
    });

  const decision =
    getPublicationDecision(
      document,
      testContext,
    );

  expect(
    decision.canBeFeatured,
  ).toBe(false);

  expect(
    decision.adsEligible,
  ).toBe(false);
});
```

## Historical

```ts
it("keeps historical documents searchable but not featured", () => {
  const document =
    createDocumentFixture({
      status: "historical",
    });

  const decision =
    getPublicationDecision(
      document,
      testContext,
    );

  expect(
    decision.includeInSearch,
  ).toBe(true);

  expect(
    decision.searchRankMultiplier,
  ).toBeLessThan(1);

  expect(
    decision.canBeFeatured,
  ).toBe(false);
});
```

## Superseded

```ts
it("keeps superseded pages readable but removes them from discovery surfaces", () => {
  const document =
    createDocumentFixture({
      status: "superseded",
      relations: {
        prerequisites: [],
        next: [],
        related: [],
        supersededBy:
          "replacement-document",
      },
    });

  const decision =
    getPublicationDecision(
      document,
      testContext,
    );

  expect(decision.render).toBe(true);
  expect(decision.index).toBe(false);

  expect(
    decision.includeInSearch,
  ).toBe(false);

  expect(
    decision.includeInSitemap,
  ).toBe(false);
});
```
