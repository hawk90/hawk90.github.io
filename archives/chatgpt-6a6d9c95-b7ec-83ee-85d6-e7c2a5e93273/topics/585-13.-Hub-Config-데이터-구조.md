# 13. Hub Config 데이터 구조

초기에는 TypeScript config 정도면 충분하다.

```ts
interface TopicHubConfig {
  id: string;
  title: string;
  description: string;

  startHere: string[];
  featured: string[];

  concepts: Array<{
    id: string;
    title: string;
    description: string;
    representativeArticle?: string;
    articles?: string[];
  }>;

  learningPaths: Array<{
    id: string;
    title: string;
    description: string;
    articles: string[];
  }>;

  debugSections: Array<{
    id: string;
    title: string;
    description: string;
    articles: string[];
  }>;

  relatedTopics: string[];
}
```

예시:

```ts
export const PCIE_CXL_HUB: TopicHubConfig = {
  id: "pcie-cxl",
  title: "PCIe & CXL",
  description:
    "PCIe 장치 초기화와 데이터 이동부터 CXL 메모리 주소 변환과 Linux 통합까지 다룹니다.",

  startHere: [
    "pcie-device-initialization",
    "pcie-bar-sizing",
    "cxl-memory-address-translation",
  ],

  featured: [
    "pcie-device-initialization",
    "pcie-bar-sizing",
    "linux-pci-enumeration",
    "pcie-msix",
    "cxl-memory-address-translation",
  ],

  concepts: [
    {
      id: "enumeration",
      title: "Enumeration & Configuration Space",
      description:
        "BDF 탐색과 Configuration Space를 통해 운영체제가 장치를 발견하는 과정입니다.",
      representativeArticle: "pcie-configuration-space",
    },
  ],

  learningPaths: [
    {
      id: "driver",
      title: "PCIe Driver 입문",
      description:
        "장치 발견부터 MMIO, 인터럽트와 DMA까지 Driver 개발 순서로 살펴봅니다.",
      articles: [
        "pcie-device-initialization",
        "pcie-configuration-space",
        "pcie-bar-sizing",
        "pcie-msix",
        "linux-dma-api",
      ],
    },
  ],

  debugSections: [],
  relatedTopics: [
    "firmware-bootloader",
    "linux-systems",
    "gpu-cuda",
  ],
};
```

실제 존재하지 않는 ID를 그대로 사용하지 말고 기존 글 ID와 대응시킨다.
