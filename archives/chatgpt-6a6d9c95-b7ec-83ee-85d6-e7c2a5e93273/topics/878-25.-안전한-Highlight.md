# 25. 안전한 Highlight

나쁜 구현:

```ts
element.innerHTML =
  text.replace(
    query,
    `<mark>${query}</mark>`,
  );
```

사용자 입력이 HTML로 삽입될 수 있다.

권장:

```ts
export function appendHighlightedText(
  container: HTMLElement,
  text: string,
  ranges: readonly MatchRange[],
): void {
  let cursor = 0;

  for (const range of ranges) {
    container.append(
      document.createTextNode(
        text.slice(
          cursor,
          range.start,
        ),
      ),
    );

    const mark =
      document.createElement("mark");

    mark.textContent =
      text.slice(
        range.start,
        range.end,
      );

    container.append(mark);

    cursor = range.end;
  }

  container.append(
    document.createTextNode(
      text.slice(cursor),
    ),
  );
}
```
