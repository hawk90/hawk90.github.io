# 55. Topic 목록 Query

```ts
export function getTopicDocuments(
  topicId: string,
  manifest: ContentManifest,
  context: PublicationContext,
): ContentDocument[] {
  return (
    manifest.byTopic.get(topicId) ??
    []
  )
    .filter((document) =>
      getPublicationDecision(
        document,
        context,
      ).includeInTopicLists
    )
    .sort(compareUpdatedDesc);
}
```
