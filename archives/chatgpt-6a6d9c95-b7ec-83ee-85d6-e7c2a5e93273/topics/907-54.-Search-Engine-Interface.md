# 54. Search Engine Interface

```ts
export interface SearchOptions {
  limit?: number;
  topicId?: string;
  type?: ContentType;
}

export interface SearchResult {
  documentId: string;
  score: number;
  matchedFields:
    readonly SearchMatchedField[];
  matchedTerms:
    readonly string[];
}

export interface SearchEngine {
  search(
    query: string,
    options?: SearchOptions,
  ): readonly SearchResult[];
}
```
