# 91. Legacy Adapter

```ts
export function adaptLegacyEntry(
  entry: LegacyArticleEntry,
): ContentDocument {
  return {
    id: deriveLegacyId(entry),
    title: entry.title,
    description:
      entry.description ??
      createFallbackDescription(entry),

    type: "unclassified",
    status: "needs-review",
    topicId: "unclassified",

    // ...
  };
}
```

하지만 `unclassified` 문서를 Featured·Start Here에 넣을 수 없게 한다.
