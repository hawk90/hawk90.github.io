# 97. Article Page Model 생성

```ts
export interface ArticlePageModel {
  document: ContentDocument;

  publication:
    PublicationDecision;

  statusPresentation:
    StatusPresentation;

  replacement?:
    ContentDocument;

  prerequisites:
    ContentDocument[];

  next:
    ContentDocument[];

  related:
    ContentDocument[];
}
```

생성:

```ts
export function buildArticlePageModel(
  document: ContentDocument,
  manifest: ContentManifest,
  context: PublicationContext,
): ArticlePageModel {
  const publication =
    getPublicationDecision(
      document,
      context,
    );

  return {
    document,
    publication,

    statusPresentation:
      getStatusPresentation(
        document,
      ),

    replacement:
      document.relations.supersededBy
        ? manifest.byId.get(
            document.relations
              .supersededBy,
          )
        : undefined,

    prerequisites:
      requireDocuments(
        document.relations
          .prerequisites,
        manifest,
      ),

    next:
      requireDocuments(
        document.relations.next,
        manifest,
      ),

    related:
      requireDocuments(
        document.relations.related,
        manifest,
      ),
  };
}
```
