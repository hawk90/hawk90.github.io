## `policy.ts`

```ts
export function getPublicationDecision(
  document: ContentDocument,
  environment: "development" | "production",
): PublicationDecision {
  if (document.status === "archived") {
    return {
      render: true,
      index: false,
      includeInSearch: false,
      includeInSitemap: false,
      includeInRss: false,
      includeInTopicLists: false,
      canBeFeatured: false,
      adsEligible: false,
    };
  }

  if (document.status === "superseded") {
    return {
      render: true,
      index: false,
      includeInSearch: false,
      includeInSitemap: false,
      includeInRss: false,
      includeInTopicLists: false,
      canBeFeatured: false,
      adsEligible: false,
    };
  }

  if (document.status === "historical") {
    return {
      render: true,
      index: true,
      includeInSearch: true,
      includeInSitemap: true,
      includeInRss: false,
      includeInTopicLists: true,
      canBeFeatured: false,
      adsEligible: false,
    };
  }

  if (document.status === "needs-review") {
    return {
      render: true,
      index: true,
      includeInSearch: true,
      includeInSitemap: true,
      includeInRss: true,
      includeInTopicLists: true,
      canBeFeatured: false,
      adsEligible: false,
    };
  }

  return {
    render: true,
    index: true,
    includeInSearch: true,
    includeInSitemap: true,
    includeInRss: true,
    includeInTopicLists: true,
    canBeFeatured: true,
    adsEligible: true,
  };
}
```

정확한 정책은 운영 결정에 따라 바꿀 수 있다.

핵심은 정책이 한 곳에 있다는 것이다.
