# 21. 검색 결과 View Model

Domain 결과를 그대로 표시하지 않는다.

```ts
export interface SearchResultViewModel {
  id: string;
  href: string;

  title: string;
  description: string;

  topicLabel: string;
  typeLabel: string;
  statusLabel?: string;

  matchedContext?: string;
  matchedField:
    | "title"
    | "alias"
    | "heading"
    | "symbol"
    | "error"
    | "description";

  updatedLabel?: string;
}
```
