# 68. ContentValidationError

export class ContentValidationError
  extends Error {
  readonly issues:

  constructor(
    issues:
      readonly ValidationIssue[],
  ) {
    super(
      formatValidationIssues(issues),
    );

    this.name =
      "ContentValidationError";
