# 75. Matrix Test

```ts
const cases = [
  {
    status: "current",
    expected: {
      index: true,
      featured: true,
    },
  },
  {
    status: "historical",
    expected: {
      index: true,
      featured: false,
    },
  },
  {
    status: "superseded",
    expected: {
      index: false,
      featured: false,
    },
  },
] as const;

it.each(cases)(
  "applies publication policy for $status",
  ({ status, expected }) => {
    const document =
      createDocumentFixture({
        status,
      });

    const decision =
      getPublicationDecision(
        document,
        testContext,
      );

    expect(decision.index)
      .toBe(expected.index);

    expect(
      decision.canBeFeatured,
    ).toBe(expected.featured);
  },
);
```
