# 52. 범용 Query 구현

```ts
export function queryDocuments(
  manifest: ContentManifest,
  context: PublicationContext,
  query: ContentQuery = {},
): ContentDocument[] {
  let results =
    manifest.documents.filter(
      (document) => {
        const decision =
          getPublicationDecision(
            document,
            context,
          );

        if (!decision.render) {
          return false;
        }

        if (
          query.topicIds &&
          !query.topicIds.includes(
            document.topicId,
          )
        ) {
          return false;
        }

        if (
          query.types &&
          !query.types.includes(
            document.type,
          )
        ) {
          return false;
        }

        if (
          query.statuses &&
          !query.statuses.includes(
            document.status,
          )
        ) {
          return false;
        }

        return true;
      },
    );

  results = sortDocuments(
    results,
    query.sort ??
      "published-desc",
  );

  if (query.limit !== undefined) {
    results = results.slice(
      0,
      query.limit,
    );
  }

  return results;
}
```
